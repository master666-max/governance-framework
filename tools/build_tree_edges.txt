# ---------- 校验 ----------
errors = []
seen = set()
for (f, t, rel, anchor, inf, reason) in E:
    if rel not in RELATIONS: errors.append(f"relation 非法: {rel}")
    if anchor not in ledger: errors.append(f"anchor 不在账本: {anchor} ({f}->{t})")
    for n in (f, t):
        if n not in nodes and n not in HUBS: errors.append(f"节点不存在: {n} ({f}->{t})")
    if inf not in ("yes", "no"): errors.append(f"INFERRED 非法: {inf}")
    if f == t: errors.append(f"自连边: {f}")
    key = (f, t, rel, anchor)
    if key in seen: errors.append(f"重复边: {key}")
    seen.add(key)
if errors:
    print("校验失败："); [print(" -", e) for e in errors]; sys.exit(1)

# 时间线纪律自检（粗粒度序：宇宙序 + SIM 宇宙内实验号序）
URANK = {"SIM":0,"DB":1,"CB":1.5,"REAL":2,"DS":3,"VERIF":3.5,"PART2":3.5,"REVIEW":4,"SYN":4,
         "GLM":3.5,"QWEN":3.5,"R2":4.5,"R2A":4.6,"PLAN":4.4,"KB":5,"META":5,"DEPLOY":5,
         "PART1":5,"L8LAB":5,"RADAR":5}
def uof(n): return n.split("-")[0] if n not in HUBS else n
def simnum(n):
    if n not in node_universe or node_universe[n] != "SIM": return None
    m = re.search(r"E(\d+)", n)
    if not m: return 9000  # SIM-SUM/SUM14/R 轮次报告等非实验号节点按大数处理
    v = int(m.group(1))
    suffix = n.split("E" + m.group(1))[-1]
    if suffix: v += 0.5          # E14C/E29b/E81c 等修正变体视为略后
    if n.startswith("SIM-WEB-"): v += 0.6
    return v
anomalies = []
# 白名单：宇宙粗秩误报（时序已在 inferred_reason 中论证）
WHITELIST = {("REAL","DS","replicates")}  # 验收报告晚于被验收的 DS 交接基线
for (f, t, rel, anchor, inf, reason) in E:
    if rel not in TIME_SAFE: continue
    if (f, t, rel) in WHITELIST: continue
    uf, ut = uof(f), uof(t)
    if uf == ut == "SIM":
        nf, nt = simnum(f), simnum(t)
        if nf is not None and nt is not None and nf < nt:
            anomalies.append((f, t, rel, anchor))
        continue
    rf, rt = URANK.get(uf, 9), URANK.get(ut, 9)
    if rf < rt: anomalies.append((f, t, rel, anchor))
