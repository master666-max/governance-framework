def norm_quote(q):
    return "".join(ch for ch in q if not ch.isspace())

def overlap(q1, q2):
    n1, n2 = norm_quote(q1), norm_quote(q2)
    if not n1 or not n2:
        return 0.0
    c1, c2 = collections.Counter(n1), collections.Counter(n2)
    inter = sum((c1 & c2).values())
    return inter / min(len(n1), len(n2))

def blind_quote(b):
    fp = os.path.join(ROOT, b["file"])

    j = text.find(b["end_anchor"], i + len(b["start_anchor"]))
    return text[i:j + len(b["end_anchor"])]

def build_bq(b):
    fp = os.path.join(ROOT, b["file"])
    with open(fp, encoding="utf-8", errors="replace") as fh:
        text = fh.readlines()[b["line"] - 1].rstrip("\r\n")

    j = text.find(b["end_anchor"], i + len(b["start_anchor"]))

bq_all = [norm_quote(blind_quote(b)) for b in B]

def best_match(i, window):
    a = A[i]
    la = int(a["location_anchor"].rsplit(":", 1)[1])
    na = norm_quote(a["verbatim_quote"])
    best = (0.0, None)
    for j, (bf, ln) in enumerate(b_line):
        if bf != a["source_file"] or abs(ln - la) > window:
            continue
        nb = bq_all[j]
        if not na or not nb:
            continue
        inter = sum((collections.Counter(na) & collections.Counter(nb)).values())
        contain = inter / min(len(na), len(nb))
        if contain > best[0]:
            best = (contain, j)
    return best

def run_v(window, contain_th):
    pairs = []
    used_a, used_b = set(), set()
    for i in range(len(A)):
        c, j = best_match(i, window)
        if j is not None and c >= contain_th:
            pairs.append((c, i, j))
    pairs.sort(reverse=True)
    for c, i, j in pairs:
        used_a.add(i); used_b.add(j)
    matched = [(i, j) for c, i, j in pairs]
    fd = sum(1 for i, j in matched if A[i]["claim_type"] != B[j]["claim_type"] or A[i]["status"] != B[j]["status"])
    A_un = len(A) - len(matched)
    return matched, used_a, used_b, A_un, fd
