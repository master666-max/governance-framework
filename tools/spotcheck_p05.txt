# -*- coding: utf-8 -*-
"""spotcheck_p05.txt — 阶段0.5 锚点抽检：取本阶段新入账断言中 claim_id
十六进制尾字符为 '7' 者（不足10条全抽），逐条回原文核对，记 audit-log.csv。"""
import os, csv, datetime

ROOT = r"D:\zcode专用！！！！危险！！！！！！！！！"
AUD = os.path.join(ROOT, "workspace-audit")
LEDGER = os.path.join(AUD, "claim_ledger.csv")

with open(LEDGER, encoding="utf-8", newline="") as f:
    rows = list(csv.DictReader(f))
# 阶段0.5 新入账 = 试金石1/2/3 的行（阶段0 的 P0 前缀在 audit-log 已查过）
p05 = [r for r in rows if not r["claim_id"].startswith(("a35e3f", "982e22", "880bac", "480945", "cafb86", "01e7e5", "c71b26", "a6c0b3", "412388", "6c40f4"))]
picked = [r for r in p05 if r["claim_id"][-1] == "7"]
if len(picked) < 10:
    picked = p05  # 不足10条全抽
print(f"阶段0.5 新入账 {len(p05)} 条，抽取 {len(picked)} 条（尾数7规则{'生效' if len(picked)>=10 else '→不足10全抽'}）")

cache = {}
results = []
for r in picked:
    src, quote = r["source_file"], r["verbatim_quote"]
    line_no = int(r["location_anchor"].rsplit(":", 1)[1])
    if src not in cache:
        with open(os.path.join(ROOT, src), encoding="utf-8", errors="replace") as f:
            cache[src] = f.readlines()
    ok = 1 <= line_no <= len(cache[src]) and quote in cache[src][line_no - 1]
    results.append((r["claim_id"], ok, src, line_no))

log_path = os.path.join(AUD, "audit-log.csv")
with open(log_path, "a", encoding="utf-8", newline="") as f:
    w = csv.writer(f)
    ts = datetime.datetime.now().isoformat(timespec="seconds")
    for cid, ok, src, ln in results:
        w.writerow([f"P05:{cid}", "命中" if ok else "未命中", f"{src}:{ln}", ts, "0.5"])

hit = sum(1 for _, ok, _, _ in results if ok)
print(f"抽检 {len(results)} 条，命中 {hit}，命中率 {hit/len(results)*100 if results else 100:.0f}%")
