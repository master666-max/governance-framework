def lines_of(rel):
    with io.open(ROOT + "\\" + rel, "r", encoding="utf-8-sig") as f:
        return f.read().splitlines()

L1, L2, L3, L4, L5, L6, L7 = (lines_of(p) for p in (S1, S2, S3, S4, S5, S6, S7))

def sha(s):
    return hashlib.sha256(s.encode("utf-8")).hexdigest()

def cut200(s):
    return s[:200]

PCT = re.compile(r"\d+(?:\.\d+)?%(?:\s*→\s*\d+(?:\.\d+)?%)?")
XNUM = re.compile(r"\d+(?:\.\d+)?x")
M35 = re.compile(r"3\.5M")
def extnums(line):
    hits = PCT.findall(line) + XNUM.findall(line) + M35.findall(line)
    seen, out = set(), []
    for h in hits:
        if h not in seen:
            seen.add(h); out.append(h)
    return "；".join(out)

MECH = {}
for ids, m in [("A1 A14 A16 A18", "A派-分层记忆OS"), ("A12 A19 A27", "B派-管线抽取"),
               ("A8 A9 A11 A15", "C派-图/联想检索"), ("A2 A3 A4 A5 A10 A23", "D派-认知架构/反思"),
               ("A6 A13 A20", "E派-生物启发/遗忘/睡眠"), ("A17", "F派-长上下文压缩"),
               ("A26 A28", "G派-产品实践"), ("A21 A22", "自进化/自指"),
               ("A29 A30 A31 A32 A33", "认知科学经典"), ("A25", "上下文工程/技术报告"),
               ("A7", "综述参照"), ("A24", "用户内部文档")]:
    for i in ids.split():
        MECH[i] = m

RL = {"A17", "A19"}  # 需权重访问例外
def tier_of(line):
    m = re.search(r"`(T\d)[^`]*`", line)
    return m.group(1) if m else None

def test_of(tier, paper):
    if paper in RL:
        return "需权重访问"
    if tier == "T1" or tier == "T2":
        return "Y"
    if tier == "T3":
        return "N"
    return ""

lit_rows = []
consumed = set()  # (filekey, lineno 1-based)

def add(paper, batch, rel, ln, quote, row_type, mech, test, status=""):
    assert quote in lines_of(rel)[ln - 1], "VERBATIM FAIL: %s:%d" % (rel, ln)
    assert len(quote) <= 200
    consumed.add((rel, ln))
    lit_rows.append({"lit_id": sha(paper + rel + ":%d" % ln + quote[:80]),
                     "paper_id": paper, "batch": batch, "source_file": rel,
                     "anchor": "%s:%d" % (rel, ln), "verbatim_quote": quote,
                     "row_type": row_type, "mechanism_type": mech,
                     "testability": test, "external_claim_numbers": extnums(lines_of(rel)[ln - 1]),
                     "status": status})

# ---- S2: 33 纸 × (简介, 落地) ----
cur = None
s2_anchor = {}
s2_tier = {}
for i, ln in enumerate(L2, 1):
    m = re.match(r"^### (A\d+) ", ln)
    if m:
        cur = m.group(1)
        continue
    if cur and ln.startswith("- **简介**："):
        add(cur, "旧33", S2, i, cut200(ln), "简介", MECH[cur], "")
        s2_anchor[cur] = "%s:%d" % (S2, i)
    elif cur and ln.startswith("- **落地**："):
        t = tier_of(ln)
        s2_tier[cur] = t
        add(cur, "旧33", S2, i, cut200(ln), "落地", MECH[cur], test_of(t, cur))
